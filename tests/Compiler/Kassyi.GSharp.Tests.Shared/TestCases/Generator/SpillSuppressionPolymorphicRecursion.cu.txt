using Kassyi.GSharp.Core;

// Generic structs whose constructors construct a struct with a type argument that holds their own type parameter
// (polymorphic recursion): every step is a new construction. Kept out of the *.cs test cases, which the IR tests
// compile and walk, because the generator itself does not handle this form yet
namespace PolymorphicRecursion
{
    public struct Wrap<T> where T : struct
    {
        public T Inner;
    }

    // Deep<T> -> Deep<Wrap<T>>
    public struct Deep<T> where T : struct
    {
        public int N;

        public Deep(int n)
        {
            N = Kernels.Step(n);
            if (n > 0)
            {
                N += new Deep<Wrap<T>>(n - 1).N;
            }
        }
    }

    // A<T> -> B<Wrap<T>> -> A<Wrap<T>>
    public struct A<T> where T : struct
    {
        public int N;

        public A(int n)
        {
            N = Kernels.Step(n);
            if (n > 0)
            {
                N += new B<Wrap<T>>(n - 1).N;
            }
        }
    }

    public struct B<T> where T : struct
    {
        public int N;

        public B(int n)
        {
            N = n;
            if (n > 0)
            {
                N += new A<Wrap<T>>(n - 1).N;
            }
        }
    }

    // S<T, U> -> S<U, Wrap<T>>
    public struct S<T, U> where T : struct where U : struct
    {
        public int N;

        public S(int n)
        {
            N = Kernels.Step(n);
            if (n > 0)
            {
                N += new S<U, Wrap<T>>(n - 1).N;
            }
        }
    }

    // M<T> -> P<Wrap<T>> -> M<U>: the expansive call and the call back form the cycle
    public struct M<T> where T : struct
    {
        public int N;

        public M(int n)
        {
            N = Kernels.Step(n);
            if (n > 0)
            {
                N += new P<Wrap<T>>(n - 1).N;
            }
        }
    }

    public struct P<U> where U : struct
    {
        public int N;

        public P(int n)
        {
            N = n;
            if (n > 0)
            {
                N += new M<U>(n - 1).N;
            }
        }
    }

    // Carried by new T(): G<T> creates T, and V<U>() creates G<V<V<U>>>, whose new T() runs V<V<U>>(), ...
    public struct G<T> where T : struct
    {
        public T Value;
        public int N;

        public G(int n)
        {
            Value = new T();
            N = Kernels.Step(n);
        }
    }

    public struct V<U> where U : struct
    {
        public int A;

        public V()
        {
            A = new G<V<V<U>>>(0).N;
        }
    }

    public static partial class Kernels
    {
        [GpuDeviceFunction]
        public static int Step(int x) => x + 1;

        [GpuKernel]
        public static void DeepKernel(int[] d)
        {
            d[0] = new Deep<int>(d[1]).N;
        }

        [GpuKernel]
        public static void AlternatingKernel(int[] d)
        {
            d[0] = new A<int>(d[1]).N;
        }

        [GpuKernel]
        public static void SwapKernel(int[] d)
        {
            d[0] = new S<int, long>(d[1]).N;
        }

        [GpuKernel]
        public static void NewTypeParameterKernel(int[] d)
        {
            d[0] = new V<int>().A;
        }

        [GpuKernel]
        public static void MutualKernel(int[] d)
        {
            d[0] = new M<int>(d[1]).N;
        }
    }
}
