using Kassyi.GSharp.Core;

namespace Kassyi.GSharp.Tests.TestCases.Analyzers;

public struct MyBigInt
{
    public int Value;
    
    [GpuDeviceFunction]
    public MyBigInt(int val)
    {
        Value = val;
        // Inner parallel logic
        Gpu.GridStrideLoop(10, i => { });
    }

    [GpuDeviceFunction]
    public static MyBigInt operator +(MyBigInt a, MyBigInt b)
    {
        Gpu.SyncGrid();
        return new MyBigInt(a.Value + b.Value);
    }
}

public static class GSH35_ParallelDependentCall_SadPath
{
    [GpuDeviceFunction]
    private static void ParallelMethod()
    {
        Gpu.GridStrideLoop(100, i => { });
    }
    
    [GpuDeviceFunction]
    private static void NestedParallelMethod()
    {
        // Indirect dependency
        ParallelMethod();
    }

    [GpuKernel]
    public static void SadKernel()
    {
        // 1. Direct invocation inside SingleThread (GSH0035)
        Gpu.SingleThread(() => 
        {
            ParallelMethod();
        });
        
        // 2. Indirect invocation inside SingleThread (GSH0035)
        Gpu.SingleThread(() => 
        {
            NestedParallelMethod();
        });

        // 3. Invocation inside manual if (GSH0035)
        if (Gpu.ThreadIndex.X == 0)
        {
            ParallelMethod();
        }
        
        var a = new MyBigInt(1);
        var b = new MyBigInt(2);
        
        Gpu.SingleThread(() => 
        {
            // 4. Object creation containing parallel logic (GSH0035)
            var c = new MyBigInt(3);
            
            // 5. Operator overload containing parallel logic (GSH0035)
            var d = a + b;
        });

        // 6. Direct invocation inside SyncGrid (GSH0035)
        Gpu.SyncGrid(() => 
        {
            ParallelMethod();
        });
    }
}
